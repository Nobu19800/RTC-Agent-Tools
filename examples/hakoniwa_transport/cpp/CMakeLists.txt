cmake_minimum_required(VERSION 3.18)
project(DroneJoystickAdapterCpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DRONE_BUILD_RTC "Build the OpenRTM executable" ON)
set(HAKONIWA_ROOT "" CACHE PATH "Hakoniwa Foundation install directory")
set(PDU_INCLUDE "${HAKONIWA_ROOT}/include/hakoniwa/pdu")
if(DRONE_BUILD_RTC)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(OPENRTM REQUIRED openrtm2)
  find_package(Threads REQUIRED)
  find_library(HAKONIWA_ASSETS assets PATHS "${HAKONIWA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_executable(DroneJoystickAdapterCpp DroneJoystickAdapter.cpp)
  target_include_directories(DroneJoystickAdapterCpp PRIVATE
    ${OPENRTM_INCLUDE_DIRS} "${HAKONIWA_ROOT}/include/hakoniwa" "${PDU_INCLUDE}")
  target_link_libraries(DroneJoystickAdapterCpp PRIVATE
    "${HAKONIWA_ASSETS}" RTC2 omniORB4 omniDynamic4 omnithread Threads::Threads)
endif()

include(CTest)
option(DRONE_COVERAGE "Instrument unit tests using Clang source coverage" OFF)
set(CLANG_PROFILE_RUNTIME "" CACHE FILEPATH "Optional path to the Clang profile runtime archive")
if(BUILD_TESTING)
  set(GTEST_SOURCE_DIR "" CACHE PATH "Optional GoogleTest source checkout")
  if(GTEST_SOURCE_DIR)
    add_subdirectory("${GTEST_SOURCE_DIR}" googletest EXCLUDE_FROM_ALL)
  else()
    find_package(GTest REQUIRED)
  endif()
  add_executable(drone_adapter_tests test_adapter.cpp)
  target_include_directories(drone_adapter_tests PRIVATE "${PDU_INCLUDE}")
  target_link_libraries(drone_adapter_tests PRIVATE GTest::gtest_main)
  if(DRONE_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      message(FATAL_ERROR "DRONE_COVERAGE requires Clang")
    endif()
    target_compile_options(drone_adapter_tests PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    if(CLANG_PROFILE_RUNTIME)
      target_link_libraries(drone_adapter_tests PRIVATE
        "-Wl,--whole-archive" "${CLANG_PROFILE_RUNTIME}" "-Wl,--no-whole-archive")
    else()
      target_link_options(drone_adapter_tests PRIVATE -fprofile-instr-generate)
    endif()
  endif()
  add_test(NAME drone_adapter_tests COMMAND drone_adapter_tests)
endif()
